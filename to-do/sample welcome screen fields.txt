        // Simple Tab Switching Logic
        function switchTab(tabId) {
            // Reset Tabs
            document.querySelectorAll('.tab-btn').forEach(btn => {
                btn.classList.remove('bg-primary/20', 'text-primary', 'border-primary/50', 'active');
                btn.classList.add('bg-surface-container-high/50', 'text-on-surface-variant', 'border-transparent');
                btn.setAttribute('aria-selected', 'false');
            });
            
            // Hide Contents
            document.querySelectorAll('.tab-content').forEach(content => {
                content.classList.add('hidden');
            });

            // Activate Selected Tab
            const activeBtn = document.getElementById(`tab-${tabId}`);
            activeBtn.classList.add('bg-primary/20', 'text-primary', 'border-primary/50', 'active');
            activeBtn.classList.remove('bg-surface-container-high/50', 'text-on-surface-variant', 'border-transparent');
            activeBtn.setAttribute('aria-selected', 'true');

            // Show Selected Content
            document.getElementById(`content-${tabId}`).classList.remove('hidden');
        }

        // D-Pad Simulation (Focus management for TV)
        // In a real TV app, this would be handled by spatial navigation libraries
        document.addEventListener('keydown', (e) => {
            const focusables = Array.from(document.querySelectorAll('.focusable, .input-focusable'));
            const currentFocus = document.activeElement;
            let currentIndex = focusables.indexOf(currentFocus);

            if (e.key === 'ArrowRight' || e.key === 'ArrowDown') {
                e.preventDefault();
                let nextIndex = currentIndex + 1;
                if (nextIndex >= focusables.length) nextIndex = 0;
                // Basic skip logic for hidden elements
                while(focusables[nextIndex].closest('.hidden')) {
                    nextIndex++;
                    if(nextIndex >= focusables.length) nextIndex = 0;
                }
                focusables[nextIndex].focus();
            } else if (e.key === 'ArrowLeft' || e.key === 'ArrowUp') {
                e.preventDefault();
                let prevIndex = currentIndex - 1;
                if (prevIndex < 0) prevIndex = focusables.length - 1;
                while(focusables[prevIndex].closest('.hidden')) {
                    prevIndex--;
                    if (prevIndex < 0) prevIndex = focusables.length - 1;
                }
                focusables[prevIndex].focus();
            }
        });

        // Set initial focus
        setTimeout(() => {
            document.getElementById('m3u-name').focus();
        }, 500);

    </script>
